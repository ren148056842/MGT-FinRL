#定义共同文本识别函数
def text_recognize(name,content):
    relation = []
    flag=False
    for i in range(len(name)):
        if type(content)!=float:
            if name[i] in content:
                relation.append(i)
                flag=True
    if flag==False:
        relation.append(-1)
    return relation
